cmake_minimum_required(VERSION 3.16)
project(etaac_offline C CXX ASM)
option(ET_DEVICE "Build only the offline device ELF" OFF)
option(ETAAC_BUILD_RUNTIME "Build SDK runner" ON)
option(ETAAC_FAST_FINITE "Aligned word bit inspection, preserving finite checks" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(DSP ../dsp/etaac_synth.c)
if(ET_DEVICE)
    list(APPEND CMAKE_MODULE_PATH /opt/et/lib/cmake/cmake-modules)
    include(DeviceUtils)
    set(LINKER_SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/../../et-kernels/src/linker.ld)
    add_riscv_executable(et_aac_offline)
    set(TARGET et_aac_offline.elf)
    target_sources(${TARGET} PRIVATE ${DSP} kernel.c ../../et-kernels/src/crt.S ../../et-kernels/src/libc.c)
    target_compile_definitions(${TARGET} PRIVATE ET_DEVICE=1 ET_WIDE_MEM=1 ETAAC_FAST_FINITE=$<BOOL:${ETAAC_FAST_FINITE}>)
    target_compile_options(${TARGET} PRIVATE -O2 -ffreestanding -fno-builtin -fno-zero-initialized-in-bss
        -mcmodel=medany -mabi=lp64f -march=rv64imf -mstrict-align -ffunction-sections -fdata-sections
        -fno-tree-vectorize -ffp-contract=off -fno-fast-math -fno-strict-aliasing
        -Wno-conversion -Wno-sign-conversion -Wno-undef -Wno-cast-align -Wno-unused-parameter
        -Wno-missing-prototypes -Wno-missing-declarations)
    target_link_options(${TARGET} PRIVATE -march=rv64imf -mabi=lp64f -ffp-contract=off
        -Wl,--defsym=BASE_ADDRESS=0x8005801000 -Wl,--entry=_start)
    add_custom_command(TARGET ${TARGET} POST_BUILD COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/../check-device.py $<TARGET_FILE:${TARGET}> /opt/et/bin/riscv64-unknown-elf-)
else()
    include(CTest)
    add_library(offline_native STATIC ${DSP} kernel.c)
    target_compile_definitions(offline_native PRIVATE ETAAC_FAST_FINITE=$<BOOL:${ETAAC_FAST_FINITE}>)
    target_compile_options(offline_native PRIVATE -O2 -Wall -Wextra -ffp-contract=off -fno-strict-aliasing)
    add_executable(offline-test tests.c)
    target_compile_options(offline-test PRIVATE -UNDEBUG -ffp-contract=off)
    target_link_libraries(offline-test PRIVATE offline_native)
    add_test(NAME offline-native COMMAND offline-test)
    set(GEN ${CMAKE_CURRENT_BINARY_DIR}/generated)
    add_custom_command(OUTPUT ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-runtime.py ${GEN}
        DEPENDS generate-runtime.py ../runtime/generate_runtime.py ../runtime/tests/runtime_test.cpp
                ../../libavcodec/et_runtime.cpp)
    add_custom_target(offline-runtime-generate DEPENDS ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp)
    add_executable(offline-runtime-test ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp)
    add_dependencies(offline-runtime-test offline-runtime-generate)
    target_compile_features(offline-runtime-test PRIVATE cxx_std_17)
    target_include_directories(offline-runtime-test PRIVATE . ../runtime/tests/mock)
    add_test(NAME offline-runtime-mock COMMAND offline-runtime-test)
    if(ETAAC_BUILD_RUNTIME)
        list(PREPEND CMAKE_PREFIX_PATH /opt/et /opt/et/lib/cmake)
        list(PREPEND CMAKE_MODULE_PATH /opt/et/lib/cmake /opt/et/lib/cmake/cmake-modules)
        find_package(runtime CONFIG REQUIRED)
        find_package(deviceLayer CONFIG REQUIRED)
        add_library(offline_runtime STATIC ${GEN}/runtime.cpp)
        add_dependencies(offline_runtime offline-runtime-generate)
        target_compile_features(offline_runtime PRIVATE cxx_std_17)
        target_include_directories(offline_runtime PRIVATE .)
        target_link_libraries(offline_runtime PRIVATE runtime::etrt_static deviceLayer::deviceLayer)
        add_executable(offline-run runner.c)
        target_compile_options(offline-run PRIVATE -O2 -Wall -Wextra)
        target_link_libraries(offline-run PRIVATE offline_runtime)
        set_target_properties(offline-run PROPERTIES LINKER_LANGUAGE CXX)
    endif()
endif()
