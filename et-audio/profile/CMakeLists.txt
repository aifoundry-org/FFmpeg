cmake_minimum_required(VERSION 3.16)
project(etaac_profile C CXX ASM)
option(ET_DEVICE "Build only ABI-3 profile device ELF" OFF)
option(ETAAC_BUILD_RUNTIME "Build SDK profile runner" ON)
option(ETAAC_FAST_FINITE "Aligned word finite inspection, exact arithmetic unchanged" OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GEN ${CMAKE_CURRENT_BINARY_DIR}/generated)
set(PROFILE_DSP ${GEN}/etaac_profile_synth.c)
add_custom_command(OUTPUT ${PROFILE_DSP}
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-profile-dsp.py
         ${CMAKE_CURRENT_SOURCE_DIR}/../dsp/etaac_synth.c ${PROFILE_DSP}
 DEPENDS generate-profile-dsp.py profile_synth.h ../dsp/etaac_synth.c ../dsp/synth.h ../dsp/etaac_tables.h)
add_custom_target(profile-dsp-generate DEPENDS ${PROFILE_DSP})
set(PROFILE_OFFLINE_TEST ${GEN}/profile_offline_tests.c)
add_custom_command(OUTPUT ${PROFILE_OFFLINE_TEST}
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-offline-tests.py
         ${CMAKE_CURRENT_SOURCE_DIR}/../offline/tests.c ${PROFILE_OFFLINE_TEST}
 DEPENDS generate-offline-tests.py ../offline/tests.c protocol.h)
add_custom_target(profile-offline-test-generate DEPENDS ${PROFILE_OFFLINE_TEST})
if(ET_DEVICE)
 list(APPEND CMAKE_MODULE_PATH /opt/et/lib/cmake/cmake-modules)
 include(DeviceUtils)
 set(LINKER_SCRIPT ${CMAKE_CURRENT_SOURCE_DIR}/../../et-kernels/src/linker.ld)
 add_riscv_executable(et_aac_profile)
 set(TARGET et_aac_profile.elf)
 target_sources(${TARGET} PRIVATE ${PROFILE_DSP} kernel.c ../../et-kernels/src/crt.S ../../et-kernels/src/libc.c)
 add_dependencies(${TARGET} profile-dsp-generate)
 target_include_directories(${TARGET} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../dsp)
 target_compile_definitions(${TARGET} PRIVATE ET_DEVICE=1 ET_WIDE_MEM=1 ETAAC_FAST_FINITE=$<BOOL:${ETAAC_FAST_FINITE}>)
 target_compile_options(${TARGET} PRIVATE -O2 -ffreestanding -fno-builtin -fno-zero-initialized-in-bss -mcmodel=medany -mabi=lp64f -march=rv64imf -mstrict-align -ffunction-sections -fdata-sections -fno-tree-vectorize -ffp-contract=off -fno-fast-math -fno-strict-aliasing -Wno-conversion -Wno-sign-conversion -Wno-undef -Wno-cast-align -Wno-unused-parameter -Wno-missing-prototypes -Wno-missing-declarations)
 target_link_options(${TARGET} PRIVATE -march=rv64imf -mabi=lp64f -ffp-contract=off -Wl,--defsym=BASE_ADDRESS=0x8005801000 -Wl,--entry=_start)
 add_custom_command(TARGET ${TARGET} POST_BUILD COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../check-device.py $<TARGET_FILE:${TARGET}> /opt/et/bin/riscv64-unknown-elf-)
else()
 include(CTest)
 add_library(profile_native STATIC ${PROFILE_DSP} kernel.c)
 add_dependencies(profile_native profile-dsp-generate)
 target_include_directories(profile_native PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../dsp)
 target_compile_definitions(profile_native PRIVATE ETAAC_FAST_FINITE=$<BOOL:${ETAAC_FAST_FINITE}>)
 target_compile_options(profile_native PRIVATE -O2 -Wall -Wextra -ffp-contract=off -fno-strict-aliasing)
 add_library(profile_reference STATIC ../dsp/etaac_synth.c)
 target_compile_definitions(profile_reference PRIVATE etaac_imdct=etaac_reference_imdct)
 target_compile_options(profile_reference PRIVATE -O2 -Wall -Wextra -ffp-contract=off -fno-strict-aliasing)
 add_executable(profile-test tests.c)
 target_compile_options(profile-test PRIVATE -UNDEBUG -ffp-contract=off)
 target_include_directories(profile-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(profile-test PRIVATE profile_native profile_reference)
 add_test(NAME profile-native COMMAND profile-test)
 add_executable(profile-offline-compat-test ${PROFILE_OFFLINE_TEST})
 add_dependencies(profile-offline-compat-test profile-offline-test-generate)
 target_compile_options(profile-offline-compat-test PRIVATE -UNDEBUG -ffp-contract=off)
 target_include_directories(profile-offline-compat-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(profile-offline-compat-test PRIVATE profile_native)
 add_test(NAME profile-offline-compat-native COMMAND profile-offline-compat-test)
 add_custom_command(OUTPUT ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-runtime.py ${GEN}
  DEPENDS generate-runtime.py ../runtime/generate_runtime.py ../runtime/tests/runtime_test.cpp ../../libavcodec/et_runtime.cpp)
 add_custom_target(profile-runtime-generate DEPENDS ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp)
 add_executable(profile-runtime-test ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp)
 add_dependencies(profile-runtime-test profile-runtime-generate)
 target_compile_features(profile-runtime-test PRIVATE cxx_std_17)
 target_include_directories(profile-runtime-test PRIVATE . ../runtime/tests/mock)
 add_test(NAME profile-runtime-mock COMMAND profile-runtime-test)
 if(ETAAC_BUILD_RUNTIME)
  list(PREPEND CMAKE_PREFIX_PATH /opt/et /opt/et/lib/cmake)
  list(PREPEND CMAKE_MODULE_PATH /opt/et/lib/cmake /opt/et/lib/cmake/cmake-modules)
  find_package(runtime CONFIG REQUIRED)
  find_package(deviceLayer CONFIG REQUIRED)
  add_library(profile_runtime STATIC ${GEN}/runtime.cpp)
  add_dependencies(profile_runtime profile-runtime-generate)
  target_compile_features(profile_runtime PRIVATE cxx_std_17)
  target_include_directories(profile_runtime PRIVATE .)
  target_link_libraries(profile_runtime PRIVATE runtime::etrt_static deviceLayer::deviceLayer)
  add_executable(profile-run runner.c)
  target_compile_options(profile-run PRIVATE -O2 -Wall -Wextra)
  target_link_libraries(profile-run PRIVATE profile_runtime)
  set_target_properties(profile-run PROPERTIES LINKER_LANGUAGE CXX)
 endif()
endif()
