cmake_minimum_required(VERSION 3.16)
project(etaac_full_host C CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
option(ETAAC_BUILD_RUNTIME "Build actual SDK host runner, never opened by CTest" ON)
set(GEN ${CMAKE_CURRENT_BINARY_DIR}/generated)
add_custom_command(OUTPUT ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp
 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/generate-runtime.py ${GEN}
 DEPENDS generate-runtime.py runtime.h protocol.h ../runtime/generate_runtime.py ../runtime/tests/runtime_test.cpp ../../libavcodec/et_runtime.cpp)
add_custom_target(full-runtime-generate DEPENDS ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp)
add_executable(full-runtime-test ${GEN}/runtime.cpp ${GEN}/runtime_test.cpp)
add_dependencies(full-runtime-test full-runtime-generate)
target_compile_features(full-runtime-test PRIVATE cxx_std_17)
target_include_directories(full-runtime-test PRIVATE . ../runtime/tests/mock)
add_test(NAME full-runtime-mock COMMAND full-runtime-test)
if(ETAAC_BUILD_RUNTIME)
 list(PREPEND CMAKE_PREFIX_PATH /opt/et /opt/et/lib/cmake)
 list(PREPEND CMAKE_MODULE_PATH /opt/et/lib/cmake /opt/et/lib/cmake/cmake-modules)
 find_package(runtime CONFIG REQUIRED)
 find_package(deviceLayer CONFIG REQUIRED)
 add_library(full_runtime STATIC ${GEN}/runtime.cpp)
 add_dependencies(full_runtime full-runtime-generate)
 target_compile_features(full_runtime PRIVATE cxx_std_17)
 target_include_directories(full_runtime PRIVATE .)
 target_link_libraries(full_runtime PRIVATE runtime::etrt_static deviceLayer::deviceLayer)
 add_executable(full-run runner.c)
 target_compile_options(full-run PRIVATE -O2 -Wall -Wextra -Werror)
 target_link_libraries(full-run PRIVATE full_runtime)
 set_target_properties(full-run PROPERTIES LINKER_LANGUAGE CXX)
 add_executable(full-emulator-run runner.c)
 target_compile_definitions(full-emulator-run PRIVATE ETAAC_FULL_EMULATOR=1)
 target_compile_options(full-emulator-run PRIVATE -O2 -Wall -Wextra -Werror)
 target_link_libraries(full-emulator-run PRIVATE full_runtime)
 set_target_properties(full-emulator-run PROPERTIES LINKER_LANGUAGE CXX)
endif()
